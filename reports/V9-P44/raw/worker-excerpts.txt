415: def _validate_reserved(run) -> float:
416:     """Raw ``reserved_active_seconds`` validated *before* any ``max`` (r8 F5).
417: 
418:     r7 normalised with ``max(0.0, float(value) or 0.0)``; ``max(0.0, nan)`` is
419:     ``0.0`` in Python, so a NaN/inf/negative reservation was silently charged as
420:     zero and the later ``isfinite`` guard could never fire.  Presence, numeric
421:     type, finiteness and sign are therefore checked first.
422:     """
423:     run_id = getattr(run, "run_id", "?")
424:     if not hasattr(run, "reserved_active_seconds"):
425:         raise LedgerStateInvalid(f"run {run_id} has no reserved_active_seconds field")
426:     value = getattr(run, "reserved_active_seconds")
427:     if value is None:
428:         raise LedgerStateInvalid(f"run {run_id} has a null reserved_active_seconds")
429:     if isinstance(value, bool) or not isinstance(value, (int, float)):
430:         raise LedgerStateInvalid(
431:             f"run {run_id} reserved_active_seconds is not numeric: {type(value).__name__}"
432:         )
433:     value = float(value)
434:     if not math.isfinite(value):
435:         raise LedgerStateInvalid(f"run {run_id} has a non-finite reserved_active_seconds {value!r}")
436:     if value < 0:
437:         raise LedgerStateInvalid(f"run {run_id} has a negative reserved_active_seconds {value!r}")
438:     return value

441: def run_conservative_wall_seconds(run, *, instant: float | None = None) -> float:
442:     """Conservative wall-clock consumption for a single run (S1 r7/r8).
443: 
444:     - For terminal runs (completed, stopped, failed): max(reserved_active, terminal_at - created_at)
445:     - For every other lifecycle state: max(reserved_active, now - created_at)
446:     - Rejects non-finite/unrepresentable/reverse timestamps and a raw
447:       reservation that is absent, non-numeric, non-finite or negative.
448:     """
449:     now = time.time() if instant is None else float(instant)
450:     if not math.isfinite(now):
451:         raise LedgerTimestampInvalid(f"accounting instant is not finite: {now!r}")
452:     run_id = getattr(run, "run_id", "?")
453:     created_at = _parse_timestamp(getattr(run, "created_at", None))
454:     updated_at = _parse_timestamp(getattr(run, "updated_at", None))
455: 
456:     if updated_at < created_at:
457:         raise LedgerTimestampInvalid(
458:             f"run {run_id} has reverse timestamps: updated {updated_at} < created {created_at}"
459:         )
460: 
461:     state = getattr(run, "state", None)
462:     if state is None:
463:         raise LedgerStateInvalid(f"run {run_id} has no state")
464:     if state not in KNOWN_STATES:
465:         raise LedgerStateInvalid(
466:             f"run {run_id} has an unknown state {state!r}; a corrupt or foreign ledger row "
467:             "is refused instead of being charged as active or as zero"
468:         )
469: 
470:     reserved_active = _validate_reserved(run)
471: 
472:     if state in TERMINAL_STATES:
473:         terminal_elapsed = max(0.0, updated_at - created_at)
474:         return max(reserved_active, terminal_elapsed)
475:     # An active row may legitimately have a clock high-water ahead of ``now``
476:     # (the RunStore keeps one); a negative age is a real zero, not a refusal.
477:     active_elapsed = max(0.0, now - created_at)
478:     return max(reserved_active, active_elapsed)

481: def phase_total_wall_seconds(
482:     runs, *, instant: float | None = None, experiment_id: str | None = None
483: ) -> float:
484:     """Total conservative wall-clock time consumed across the phase's runs.
485: 
486:     r11: ``experiment_id`` scopes the total to one registration, so the new
487:     experiment's 9 x 1200 s budget is measured against its own runs only (the
488:     old experiment's ~1.2 h stays recorded but is not charged to it).  ``None``
489:     sums every row, the pre-r11 behaviour.
490:     """
491:     rows = runs
492:     if experiment_id is not None:
493:         rows = [run for run in runs if getattr(run, "experiment_id", None) == experiment_id]
494:     return sum(run_conservative_wall_seconds(r, instant=instant) for r in rows)

497: def phase_ledger_breakdown(
498:     runs, *, instant: float | None = None, experiment_id: str | None = None
499: ) -> dict:
500:     """Per-row conservative accounting, for evidence and refusals (S1 r7/r8).
501: 
502:     Each row reports the two candidate quantities separately so a reader can see
503:     that the terminal interval and the reserved allowance are ``max``-ed per row
504:     and then summed -- never ``max`` of two global totals, and never the
505:     calendar age of a terminal row.
506: 
507:     r11: ``experiment_id`` scopes the breakdown to one registration, matching
508:     :func:`phase_total_wall_seconds`; ``None`` reports every row.
509:     """
510:     now = time.time() if instant is None else float(instant)
511:     if experiment_id is not None:
512:         runs = [run for run in runs if getattr(run, "experiment_id", None) == experiment_id]
513:     rows = []
514:     total = 0.0
515:     for run in runs:
516:         counted = run_conservative_wall_seconds(run, instant=now)
517:         state = getattr(run, "state", None)
518:         reserved = _validate_reserved(run)
519:         created_at = _parse_timestamp(run.created_at)
520:         updated_at = _parse_timestamp(run.updated_at)
521:         terminal_interval = max(0.0, updated_at - created_at)
522:         active_age = max(0.0, now - created_at)
523:         rows.append(
524:             {
525:                 "run_id": getattr(run, "run_id", None),
526:                 "state": state,
527:                 "reserved_active_seconds": reserved,
528:                 "terminal_interval_seconds": terminal_interval if state in TERMINAL_STATES else None,
529:                 "active_age_seconds": active_age if state in ACTIVE_STATES else None,
530:                 "counted_seconds": counted,
531:                 "rule": "max(reserved, terminal_interval)" if state in TERMINAL_STATES
532:                 else "max(reserved, now-created)",
533:             }
534:         )
