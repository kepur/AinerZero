def score_pool(pool, nets):
    """Per sample: (disagreement, true error of the ensemble mean), each the max over disks."""
    dis, err = {}, {}
    for n, e in pool.by_n.items():
        p, v, pil = mx.array(e["p"]), mx.array(e["v"]), mx.array(e["pil"])
        preds = []
        for net in nets:
            _, vv, _, _ = net(p, v, pil)
            preds.append(np.array(vv, dtype=np.float64))
        st = np.stack(preds)
        mean = st.mean(0)
        dis[n] = np.linalg.norm(st - mean[None], axis=-1).mean(0).max(-1)
        err[n] = np.linalg.norm(mean - e["v1"], axis=-1).max(-1)
    return dis, err

def evaluate(models, test_eps):
    """Per model: mean velocity error per kind, and D on collision disk-steps."""
    out = {}
    pre = []
    for ep in test_eps:
        T = len(ep.steps) - 1
        P = np.stack([s.positions for s in ep.steps[:T]]).astype(np.float32)
        V = np.stack([s.velocities for s in ep.steps[:T]]).astype(np.float32)
        tgt = np.stack([s.velocities for s in ep.steps[1:T + 1]])
        kinds = np.stack([classify(ep, t) for t in range(T)])
        _, vb = Mo.bounce(P.astype(np.float64), V.astype(np.float64), ep.pillars)
        pil = mx.array(np.broadcast_to(np.array(ep.pillars, dtype=np.float32), (T, L.N_PILLARS, 3)))
        pre.append((P, V, tgt, kinds, np.linalg.norm(vb - tgt, axis=-1), pil))
    for name, m in models.items():
        sums = np.zeros(4); cnt = np.zeros(4); eb_coll = 0.0
        for P, V, tgt, kinds, eb_err, pil in pre:
            _, v, _, _ = m(mx.array(P), mx.array(V), pil)
            e = np.linalg.norm(np.array(v, dtype=np.float64) - tgt, axis=-1)
            for k in range(4):
                sums[k] += e[kinds == k].sum(); cnt[k] += (kinds == k).sum()
            eb_coll += eb_err[kinds == 0].sum()
        mean = sums / cnt
        out[name] = {"collision": mean[0], "graze": mean[1], "wall_pillar": mean[2], "free": mean[3],
                     "all": sums.sum() / cnt.sum(), "D": 1 - sums[0] / eb_coll}
    return out

def geo_gate(P: np.ndarray, V: np.ndarray) -> np.ndarray:
    """[M, n] bool for member states P, V [M, n, 2]: the disk is in a pair that the phase-1 contact
    baseline flags (approaching and touching within DT on straight lines, or overlapping and approaching)."""
    d = P[:, None, :, :] - P[:, :, None, :]                  # d[m, i, j] = p_j - p_i
    u = V[:, None, :, :] - V[:, :, None, :]
    a = (u * u).sum(-1)
    b = 2 * (d * u).sum(-1)
    c = (d * d).sum(-1) - (2 * W.RADIUS) ** 2
    disc = b * b - 4 * a * c
    with np.errstate(invalid="ignore", divide="ignore"):
        tau = (-b - np.sqrt(np.maximum(disc, 0.0))) / (2 * np.where(a > 0, a, 1.0))
    hit = ((c <= 0) & (b < 0)) | ((a > 0) & (disc >= 0) & (b < 0) & (tau >= 0) & (tau < W.DT))
    hit &= ~np.eye(P.shape[1], dtype=bool)[None]
    return hit.any(-1)

def make_pool(episodes: Sequence, n_samples: int, rng: np.random.Generator, physics, *, exact_frac: float = EXACT_FRAC) -> Pool:
    """Draw states from the episodes, perturb them, label with ``physics`` rolled out 1, 5 and 10 steps.

    Step 1 runs per sample (the simulator logs contacts of its first copy only); steps 2-10 run
    batched over the samples of one episode with one disk count."""
    draws = []
    for _ in range(n_samples):
        e = int(rng.integers(len(episodes)))
        s = episodes[e].steps[int(rng.integers(len(episodes[e].steps)))]
        pos, vel = perturb(s.positions, s.velocities, rng, exact=rng.random() < exact_frac)
        draws.append((e, pos, vel))
    groups: dict[tuple[int, int], list[int]] = {}
    for i, (e, pos, _) in enumerate(draws):
        groups.setdefault((len(pos), e), []).append(i)
    rows: dict[int, dict[str, list]] = {}
    for (n, e), idx in sorted(groups.items()):
        pillars = tuple(map(tuple, episodes[e].pillars))
        P = np.stack([draws[i][1] for i in idx]); V = np.stack([draws[i][2] for i in idx])

def rollout_loss(model: Dynamics, batch: dict[str, mx.array]) -> mx.array:
    """MSE of the model's own rollout against the simulator at 1, 5 and 10 steps, plus the contact loss."""
    p, v, pil = batch["p"], batch["v"], batch["pil"]
    loss = mx.array(0.0)
    for k in range(1, max(ROLLOUT) + 1):
        p, v, logits, _ = model(p, v, pil)
        if k == 1 and logits is not None:
            n = p.shape[1]
            off = 1 - mx.eye(n)[None]
            bce = nn.losses.binary_cross_entropy(logits, batch["contact"], with_logits=True, reduction="none")
            loss = loss + CONTACT_WEIGHT * (bce * off).sum() / mx.maximum(off.sum() * p.shape[0], 1)
        if k in ROLLOUT:
            w = ROLLOUT_WEIGHTS[ROLLOUT.index(k)]
            loss = loss + w * ((((p - batch[f"p{k}"]) / P_SCALE) ** 2).mean() + ((v - batch[f"v{k}"]) ** 2).mean())
    return loss

def sample_kinds(e: dict[str, np.ndarray]) -> tuple[np.ndarray, np.ndarray]:
    """Indices of a mean bucket's samples (contact, near miss): a disk contact at step 1, or none
    and some pair of disks ending step 1 with centres closer than NEAR_GAP. Read from the
    simulator's labels only."""
    if "contact" not in e:
        return np.zeros(0, np.int64), np.zeros(0, np.int64)
    hit = e["contact"].sum((1, 2)) > 0
    p1 = e["p1"]
    d = np.linalg.norm(p1[:, :, None] - p1[:, None], axis=-1) + np.eye(p1.shape[1])[None] * 99.0
    near = (d.min((1, 2)) < NEAR_GAP) & ~hit
    return np.flatnonzero(hit), np.flatnonzero(near)


def batches(pool: Pool, steps: int, batch: int, rng: np.random.Generator, *, coll_share: float = 0.0,
            near_share: float = 0.0) -> Iterator[tuple[int, np.ndarray]]:
    """Bucket chosen in proportion to its size, then ``batch`` indices drawn with replacement. With
    ``coll_share`` > 0 that share of the batch comes from the bucket's contact samples, with
    ``near_share`` > 0 that share from its near misses (see ``sample_kinds``); a kind the bucket
    does not have is drawn from the whole bucket instead. Order in the batch: contact, near, rest."""
    ns = sorted(pool.by_n)
    sizes = np.array([len(pool.by_n[n]["p"]) for n in ns], dtype=np.float64)
    w = sizes / sizes.sum()
    kinds = {n: sample_kinds(pool.by_n[n]) for n in ns}
    k, q = int(round(batch * coll_share)), int(round(batch * near_share))
    if k + q > batch:
        raise ValueError("the contact and near-miss shares exceed the batch")
    for _ in range(steps):
        j = int(rng.choice(len(ns), p=w))
        h, nr = kinds[ns[j]]
        parts = []
        if k and len(h):
            parts.append(h[rng.integers(0, len(h), k)])
        if q and len(nr):
            parts.append(nr[rng.integers(0, len(nr), q)])
        parts.append(rng.integers(0, int(sizes[j]), batch - sum(len(x) for x in parts)))
        yield ns[j], np.concatenate(parts)

# --- bootstrap over worlds --------------------------------------------------------------------
def bootstrap(stat: Callable[..., float], sums: dict[str, np.ndarray], *, n_boot: int = BOOT_N, seed: int = BOOT_SEED,
              m: int = 1) -> dict[str, Any]:
    """``sums`` maps names to per-world arrays (same length); ``stat`` takes their totals as keywords.

    Worlds are resampled with replacement; returns the point value, the 95% interval and a
    Bonferroni interval over ``m`` comparisons."""
    names = list(sums)
    arr = np.stack([np.asarray(sums[k], dtype=np.float64) for k in names])  # [K, W]
    n_w = arr.shape[1]
    if n_w == 0:
        return {"value": None, "ci_low": None, "ci_high": None, "bonf_low": None, "bonf_high": None, "n_worlds": 0}
    with np.errstate(divide="ignore", invalid="ignore"):
        point = float(stat(**dict(zip(names, arr.sum(1)))))
    rng = np.random.default_rng(seed)
    draws = rng.integers(0, n_w, size=(n_boot, n_w))
    tot = arr[:, draws].sum(-1)                                              # [K, n_boot]
    with np.errstate(divide="ignore", invalid="ignore"):
        boot = np.array(stat(**dict(zip(names, tot))), dtype=np.float64)
    boot = boot[np.isfinite(boot)]
    if not np.isfinite(point) or len(boot) < n_boot:
        # An undefined ratio (e.g. E-P no better than E-B) in the point value or in any resample:
        # dropping those resamples would narrow the interval, so no interval is given and every
        # gate that needs one fails. The point value is kept when it is defined.
        return {"value": float(point) if np.isfinite(point) else None, "ci_low": None, "ci_high": None,
                "bonf_low": None, "bonf_high": None, "n_worlds": int(n_w), "n_boot_finite": int(len(boot))}
    tail = (1 - LEVEL) / 2
    lo, hi = np.quantile(boot, [tail, 1 - tail])
    blo, bhi = np.quantile(boot, [tail / m, 1 - tail / m])
    return {"value": float(point), "ci_low": float(lo), "ci_high": float(hi), "bonf_low": float(blo), "bonf_high": float(bhi),
            "n_worlds": int(n_w), "n_boot_finite": int(len(boot))}


def _ratio(num, den):
    """num / den, undefined (NaN) wherever den is not positive: a resample in which a denominator
    vanishes or changes sign gives no ratio, so the bootstrap gives no interval."""
    num, den = np.asarray(num, dtype=np.float64), np.asarray(den, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore"):
        return np.where(den > 0, num / np.where(den > 0, den, 1.0), np.nan)


def stat_D(x_coll, b_coll, **_):
    """D(X) = 1 - err(X) / err(E-B) on collision disk-steps (same disk-steps for both)."""
    return 1 - _ratio(x_coll, b_coll)
