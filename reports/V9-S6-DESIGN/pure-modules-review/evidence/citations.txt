    72	def t_cdf(x: float, df: int) -> float:
    73	    """Student t CDF for a positive integer number of degrees of freedom (closed-form series)."""
    74	    if df < 1 or int(df) != df:
    75	        raise ValueError("df must be a positive integer")
    76	    th = math.atan(x / math.sqrt(df))
    77	    c2, s = math.cos(th) ** 2, math.sin(th)
    78	    if df % 2 == 1:
    79	        acc, term = 0.0, math.cos(th)
    80	        if df > 1:
    81	            acc = term
    82	            for j in range(1, (df - 3) // 2 + 1):
    83	                term *= c2 * (2 * j) / (2 * j + 1)
    84	                acc += term
    85	        return 0.5 + (th + s * acc) / math.pi if df > 1 else 0.5 + th / math.pi
    86	    acc, term = 1.0, 1.0
    87	    for j in range(1, (df - 2) // 2 + 1):
    88	        term *= c2 * (2 * j - 1) / (2 * j)
    89	        acc += term
    90	    return 0.5 + 0.5 * s * acc
