def _canon(remaining, k: int) -> np.ndarray:
    rem = np.asarray(remaining)
    if rem.ndim != 1:
        raise ValueError("remaining must be one-dimensional")
    if rem.size and not np.issubdtype(rem.dtype, np.integer):
        raise TypeError("remaining must hold integer ids")
    rem = rem.astype(np.int64)
    srt = np.unique(rem)
    if len(srt) != len(rem):
        raise ValueError("remaining has duplicate ids")
    if not 0 <= k <= len(srt):
        raise ValueError(f"cannot pick {k} from {len(srt)} remaining")
    return srt

def geo_gate(P: np.ndarray, V: np.ndarray) -> np.ndarray:
    """[M, n] bool for member states P, V [M, n, 2]: the disk is in a pair that the phase-1 contact
    baseline flags (approaching and touching within DT on straight lines, or overlapping and approaching)."""
    d = P[:, None, :, :] - P[:, :, None, :]                  # d[m, i, j] = p_j - p_i
    u = V[:, None, :, :] - V[:, :, None, :]
    a = (u * u).sum(-1)
    b = 2 * (d * u).sum(-1)
    c = (d * d).sum(-1) - (2 * W.RADIUS) ** 2
    disc = b * b - 4 * a * c
    with np.errstate(invalid="ignore", divide="ignore"):
        tau = (-b - np.sqrt(np.maximum(disc, 0.0))) / (2 * np.where(a > 0, a, 1.0))
    hit = ((c <= 0) & (b < 0)) | ((a > 0) & (disc >= 0) & (b < 0) & (tau >= 0) & (tau < W.DT))
    hit &= ~np.eye(P.shape[1], dtype=bool)[None]
    return hit.any(-1)
