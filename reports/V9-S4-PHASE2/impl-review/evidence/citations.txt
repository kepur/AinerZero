30:def geo_gate(P: np.ndarray, V: np.ndarray) -> np.ndarray:
31:    """[M, n] bool for member states P, V [M, n, 2]: the disk is in a pair that the phase-1 contact
32:    baseline flags (approaching and touching within DT on straight lines, or overlapping and approaching)."""
33:    d = P[:, None, :, :] - P[:, :, None, :]                  # d[m, i, j] = p_j - p_i
34:    u = V[:, None, :, :] - V[:, :, None, :]
35:    a = (u * u).sum(-1)
36:    b = 2 * (d * u).sum(-1)
37:    c = (d * d).sum(-1) - (2 * W.RADIUS) ** 2
38:    disc = b * b - 4 * a * c
39:    with np.errstate(invalid="ignore", divide="ignore"):
40:        tau = (-b - np.sqrt(np.maximum(disc, 0.0))) / (2 * np.where(a > 0, a, 1.0))
41:    hit = ((c <= 0) & (b < 0)) | ((a > 0) & (disc >= 0) & (b < 0) & (tau >= 0) & (tau < W.DT))
42:    hit &= ~np.eye(P.shape[1], dtype=bool)[None]
43:    return hit.any(-1)

47:    """[M, n] bool: some other disk within MSG_RADIUS with contact probability above ``p``."""
48:    if not 0.0 < p < 1.0:
49:        raise ValueError("the gate threshold must lie in (0, 1)")
50:    n = P.shape[1]
51:    dist = np.linalg.norm(P[:, :, None] - P[:, None], axis=-1)
52:    near = (dist < L.MSG_RADIUS) & ~np.eye(n, dtype=bool)[None]

321:def cv_contact_pairs(p: np.ndarray, v: np.ndarray) -> set[tuple[int, int]]:
322:    """Baseline: a pair approaching on straight lines that touches within one step."""
323:    out = set()
324:    n = len(p)
325:    for i in range(n):
326:        for j in range(i + 1, n):
327:            d, u = p[j] - p[i], v[j] - v[i]
328:            a, b, c = u @ u, 2 * d @ u, d @ d - (2 * W.RADIUS) ** 2
329:            if c <= 0 and b < 0:
330:                out.add((i, j))
331:                continue
332:            disc = b * b - 4 * a * c
333:            if a > 0 and disc >= 0 and b < 0:
334:                tau = (-b - np.sqrt(disc)) / (2 * a)
335:                if 0 <= tau < W.DT:
336:                    out.add((i, j))
