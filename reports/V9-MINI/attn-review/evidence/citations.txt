   146	    """(N, T+1, D) → (N, T+1) 整数编号：观察完全相同的帧编号相同（对原始观察的精确比对）。"""

   155	def hindsight_labels(wid: np.ndarray, gid: np.ndarray):
   156	    """wid: (B, L) 窗口内各帧的观察编号；gid: (B, G) 目标的观察编号。
   157	
   158	    返回（都是 (B, G, L−1)，对应位置 t = 0..L−2 上执行 a_t 之后）：
   159	    - ub：t 之后第一次看到目标画面用掉的步数（看不到为 inf），是代价的上界；
   160	    - done：下一帧就是目标画面；
   161	    - valid：t 这一帧还不是目标画面（已经到了的帧不参与训练，测试时也不会在那里决策）。
   162	    """
   163	    B, L = wid.shape
   164	    eq = wid[:, None, :] == gid[:, :, None]                                  # (B, G, L)
   165	    pos = np.where(eq, np.arange(L)[None, None], np.iinfo(np.int64).max)
   166	    nxt = np.minimum.accumulate(pos[..., ::-1], axis=-1)[..., ::-1]          # nxt[t] = t 及之后第一次出现
   167	    nxt_after = nxt[..., 1:]                                                 # t+1 及之后
   168	    t = np.arange(L - 1)[None, None]
   169	    ub = np.where(nxt_after < np.iinfo(np.int64).max, nxt_after - t, np.inf).astype(np.float64)
   170	    return ub, eq[..., 1:], ~eq[..., :-1]
   171

   173	def compose_target(v_next: np.ndarray, done: np.ndarray, ub: np.ndarray, cap: float) -> np.ndarray:
   174	    """y = min(下一帧是目标 ? 1 : 1 + min_a' Q⁻(t+1), 上界, cap)。"""
   175	    y = np.where(done, 1.0, 1.0 + v_next)
   176	    return np.minimum(np.minimum(y, ub), cap)

   179	def sample_batch(ds: Dataset, ids: np.ndarray, rng: np.random.Generator, B: int, G: int, L: int):
   180	    T1 = ds.obs.shape[1]
   181	    if L > T1:
   182	        raise ValueError("窗口比游走长")
   183	    i = rng.integers(0, ds.n_traj, B)
   184	    s = rng.integers(0, T1 - L + 1, B)
   185	    idx = s[:, None] + np.arange(L)[None]
   186	    obs = ds.obs[i[:, None], idx]                                            # (B, L, D)
   187	    acts = ds.act[i[:, None], idx[:, :-1]]                                   # (B, L−1)，a_t
   188	    prev = np.concatenate([np.full((B, 1), NONE_ACTION), acts], axis=1)      # 窗口第一帧没有上一步动作
   189	    wid = ids[i[:, None], idx]
   190	    gpos = rng.integers(0, L, (B, G))
   191	    og = obs[np.arange(B)[:, None], gpos]                                     # (B, G, D)
   192	    gid = wid[np.arange(B)[:, None], gpos]
   193	    return obs, acts, prev, og, wid, gid

   196	def train_attn(ds: Dataset, cfg: AttnConfig, log=print, on_step=None) -> tuple[AttnNets, dict]:
   197	    """ds 只读 obs 和 act。on_step(step, net)：每步结束后调用（探索时存检查点用）。"""
   198	    mx.random.seed(cfg.seed)
   199	    rng = np.random.default_rng(cfg.seed + 9_191)
   200	    D_ = ds.obs.shape[-1]
   201	    kw = dict(d=cfg.d, layers=cfg.layers, heads=cfg.heads, scale=cfg.scale, q_init=cfg.q_init)
   202	    net, tgt = AttnNets(D_, **kw), AttnNets(D_, **kw)
   203	    mx.eval(net.parameters())
   204	    tgt.update(tree_map(lambda p: mx.array(p), net.parameters()))
   205	    mx.eval(tgt.parameters())
   206	    opt = optim.AdamW(learning_rate=cfg.lr, weight_decay=1e-4)
   207	    ids = obs_ids(ds.obs)
   208	
   209	    def loss_fn(m, obs, prev, og, acts, y, mask):
   210	        h, x = m.encode(obs, prev)
   211	        q = m.q_values(h, x, og)[:, :, :-1]                                  # (B, G, L−1, A)
   212	        a = mx.broadcast_to(acts[:, None, :, None], q.shape[:-1] + (1,))
   213	        qa = mx.take_along_axis(q, a, axis=-1)[..., 0]
   214	        err = nn.losses.huber_loss(qa / cfg.scale, y / cfg.scale, reduction="none")
   215	        return mx.sum(err * mask) / mx.maximum(mx.sum(mask), 1.0), mx.sum(qa * mask) / mx.maximum(mx.sum(mask), 1.0)
   216	
   217	    vg = nn.value_and_grad(net, loss_fn)
   218	    hist = []
   219	    t0 = time.time()
   220	    for step in range(1, cfg.steps + 1):
   221	        obs, acts, prev, og, wid, gid = sample_batch(ds, ids, rng, cfg.batch, cfg.goals, cfg.window)
   222	        ub, done, valid = hindsight_labels(wid, gid)
   223	        o_m, p_m, g_m = mx.array(obs), mx.array(prev.astype(np.int32)), mx.array(og)
   224	        ht, xt = tgt.encode(o_m, p_m)
   225	        v_next = np.array(mx.min(tgt.q_values(ht, xt, g_m)[:, :, 1:], axis=-1))
   226	        y = compose_target(v_next, done, ub, cfg.cap)
   227	        (loss, qmean), g = vg(net, o_m, p_m, g_m, mx.array(acts.astype(np.int32)), mx.array(y.astype(np.float32)),
   228	                              mx.array(valid.astype(np.float32)))
   229	        opt.update(net, g)
   230	        tgt.update(tree_map(lambda a, b: (1.0 - cfg.tau) * a + cfg.tau * b, tgt.parameters(), net.parameters()))
   231	        mx.eval(net.parameters(), tgt.parameters(), opt.state, loss, qmean)
   232	        if step % cfg.log_every == 0 or step == 1:
   233	            m = valid
   234	            row = {"step": step, "sec": round(time.time() - t0, 1), "loss": float(loss), "q_mean": float(qmean),
   235	                   "y_mean": float(y[m].mean()), "ub_binds": float((ub[m] <= np.where(done, 1.0, 1.0 + v_next)[m]).mean()),
   236	                   "ub_known": float(np.isfinite(ub[m]).mean())}
   237	            hist.append(row)
   238	            log("  attn " + "  ".join(f"{k}={v:.4g}" if isinstance(v, float) else f"{k}={v}" for k, v in row.items()))
   239	        if on_step is not None:
   240	            on_step(step, net)
   241	    return net, {"config": asdict(cfg), "params": n_params(net), "train_sec": round(time.time() - t0, 1),
   242	                 "data": {"n_traj": int(ds.n_traj), "horizon": int(ds.horizon)}, "history": hist}

   247	class AttnAgent:
   248	    def __init__(self, nets: AttnNets, eps: float = 0.05, seed: int = 0):
   249	        self.nets, self.eps = nets, eps
   250	        self.rng = random.Random(seed)
   251	
   252	    def reset(self, ep) -> None:
   253	        self.obs = [np.asarray(o, dtype=np.float32) for o in ep.prefix_obs]
   254	        self.prev = [NONE_ACTION] + [int(a) for a in ep.prefix_acts]
   255	        self.og = mx.array(np.asarray(ep.goal_obs, dtype=np.float32)[None, None])
   256	        self.first = True
   257	        self.last = None
   258	
   259	    def q_last(self) -> np.ndarray:
   260	        h, x = self.nets.encode(mx.array(np.stack(self.obs)[None]), mx.array(np.array([self.prev], dtype=np.int32)))
   261	        return np.array(self.nets.q_values(h, x, self.og, last_only=True)).reshape(-1)
   262	
   263	    def act(self, obs: np.ndarray, view, head) -> int:
   264	        if not self.first:          # 前缀最后一帧就是任务第一帧，已经在历史里
   265	            self.obs.append(np.asarray(obs, dtype=np.float32))
   266	            self.prev.append(self.last)
   267	        self.first = False
   268	        if self.rng.random() < self.eps:
   269	            a = self.rng.randrange(W.N_ACTIONS)
   270	        else:
   271	            q = self.q_last() + 1e-6 * np.array([self.rng.random() for _ in range(W.N_ACTIONS)])
   272	            a = int(np.argmin(q))
   273	        self.last = a
   274	        return a

    64	class Block(nn.Module):
    65	    def __init__(self, d: int, heads: int):
    66	        super().__init__()
    67	        self.heads = heads
    68	        self.ln1, self.ln2 = nn.LayerNorm(d), nn.LayerNorm(d)
    69	        self.qkv = nn.Linear(d, 3 * d, bias=False)
    70	        self.o = nn.Linear(d, d, bias=False)
    71	        self.ff = mlp([d, 4 * d, d])
    72	        self.rope = nn.RoPE(d // heads)
    73	
    74	    def __call__(self, x: mx.array) -> mx.array:
    75	        B, L, d = x.shape
    76	        H, dh = self.heads, d // self.heads
    77	        q, k, v = mx.split(self.qkv(self.ln1(x)), 3, axis=-1)
    78	        q, k, v = (t.reshape(B, L, H, dh).transpose(0, 2, 1, 3) for t in (q, k, v))
    79	        q, k = self.rope(q), self.rope(k)
    80	        a = mx.fast.scaled_dot_product_attention(q, k, v, scale=dh ** -0.5, mask="causal")
    81	        x = x + self.o(a.transpose(0, 2, 1, 3).reshape(B, L, d))
    82	        return x + self.ff(self.ln2(x))

--- attn.py: q_values (GQA 分组, RoPE offset) ---
   111	    def q_values(self, h: mx.array, x: mx.array, og: mx.array, last_only: bool = False) -> mx.array:
   112	        """og: (B, G, D) 目标观察。返回各动作的代价 (B, G, L, A)；last_only 时只算最后一个位置 (B, G, 1, A)。"""
   113	        B, L, d = h.shape
   114	        G = og.shape[1]
   115	        H, dh = self.heads, d // self.heads
   116	        eg = self.goal_enc(og)                                               # (B, G, d)
   117	        hq = h[:, -1:] if last_only else h
   118	        Lq = hq.shape[1]
   119	        hb = mx.broadcast_to(hq[:, None], (B, G, Lq, d))
   120	        gb = mx.broadcast_to(eg[:, :, None], (B, G, Lq, d))
   121	        q = self.rq(mx.concatenate([hb, gb], axis=-1))                       # (B, G, Lq, d)
   122	        q = q.reshape(B, G, Lq, H, dh).transpose(0, 3, 1, 2, 4).reshape(B, H * G, Lq, dh)   # 查询头按 [头, 目标] 排，对应 GQA 的分组
   123	        kv = mx.concatenate([h, x], axis=-1)
   124	        k = self.rk(kv).reshape(B, L, H, dh).transpose(0, 2, 1, 3)
   125	        v = self.rv(kv).reshape(B, L, H, dh).transpose(0, 2, 1, 3)
   126	        k = self.rope(k)
   127	        if last_only:
   128	            q = self.rope(q, offset=L - 1)
   129	            r = mx.fast.scaled_dot_product_attention(q, k, v, scale=dh ** -0.5)
   130	        else:
   131	            q = self.rope(q)
   132	            r = mx.fast.scaled_dot_product_attention(q, k, v, scale=dh ** -0.5, mask="causal")
   133	        r = r.reshape(B, H, G, Lq, dh).transpose(0, 2, 3, 1, 4).reshape(B, G, Lq, d)
   134	        out = self.head(mx.concatenate([hb, gb, self.ro(r)], axis=-1))
   135	        return 1.0 + self.scale * nn.softplus(out + self.b0)

   476	## 11. 注意力记忆组 latent_attn（2026-11-18 起，探索，不算证据）
   477	
   478	用户 11-18 选了「先加 Transformer 组」，原因是确认段每个里程碑只登记一次，对照组要在登记前定下来。§10.5 第 5 条列的第一类强对照，就是用注意力读整段前缀的长记忆。
   479	
   480	### 11.1 做法（`src/ainerzero/v9/mini/attn.py`）
   481	
   482	这是目标条件的离线 Q 学习：目标用 hindsight 重标记（Kaelbling 1993；HER，Andrychowicz 等 2017），记忆是因果 Transformer，读完整历史（前缀加本回合已走的步）。全部自监督。
   483	
   484	**输入：** 只有观察（含指南针朝向）和智能体自己的上一步动作。不读真值位姿，也不读真值距离。
   485	
   486	**「到达」的定义：** 当前观察与目标画面完全相同。这和 latent_topo 的「同一处」一样，是对原始观察的比对。
   487	
   488	**训练目标（代价按步数计）：**
   489	- 自举：Q(h_t, a_t, g) = 1 + min_a' Q⁻(h_{t+1}, a', g)；下一帧就是目标画面时为 1。Q⁻ 是 Polyak 平均的目标网络。
   490	- 蒙特卡洛上界：同一条游走里，之后第一次看到目标画面用了多少步。
   491	- 目标取 min(自举, 上界, 200)。
   492	
   493	**结构：**
   494	- 观察编码加上一步动作的嵌入，进 4 层因果 Transformer（d=128，4 头，RoPE），得到 h_t。
   495	- 目标读出：位置 t 用 [h_t, 目标嵌入] 作查询，对 ≤ t 的所有 [h_j, 第 j 帧的输入嵌入] 做一层多头注意力，再接 MLP 输出 3 个动作的代价。这一层让模型能在记忆里找到目标画面出现在哪里、之后是怎么走的。
   496	
   497	**训练数据：**
   498	- 训练图和环境步数都与其他组相同（200 张图，每张 800 步）。
   499	- 只是把每张图的 4 条 200 步游走换成 1 条 800 步游走，训练窗口（600 帧）才覆盖得到测试时 401 帧前缀加回合步数的长度。
   500

   539	### 11.3 读法（种子 0，待 3 种子批次确认）
   540	
   541	1. **它学到了东西，但不够用。**
   542	   - 第一步选对的比例高于随机猜中，代价与 d* 正相关。
   543	   - 但成功率在随机水平附近：短目标 0.10–0.25，长目标 0–0.025。
   544	2. **加步数没用。** 3000、6000、12000 三个检查点上下跳，60 个回合的标准误约 0.05，看不出趋势。
   545	3. **不是过拟合。** 训练图上同样只有这个水平。瓶颈在学习问题本身：200 张图、16 万环境步，要学会从历史里推出几何。
   546	4. **和 §10 合起来读。**
   547	   - 三类学到的记忆都差得远：循环信念（gru_belief）、拓扑记忆（latent_topo）、注意力（latent_attn）。
   548	   - 拓扑记忆最好，因为它把「沿走过的路重放」写进了结构。
   549	   - 注意力记忆要从数据里自己学出这一点，加上几何，在这个数据量下学不出来。
