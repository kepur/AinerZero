    89	def sample_reach(ds: Dataset, rng: np.random.Generator, b: int, k_pos: int, k_neg: int):
    90	    """返回 (i, t1, t2, y)：同一条轨迹 i 上的两帧及标签（1 = 相隔 ≤ k_pos，0 = 相隔 ≥ k_neg）。"""
    91	    T = ds.horizon
    92	    if k_neg > T:
    93	        raise ValueError("k_neg 超过轨迹长度")
    94	    i = rng.integers(0, ds.n_traj, b)
    95	    t1 = rng.integers(0, T + 1, b)
    96	    y = (rng.random(b) < 0.5).astype(np.float32)
    97	    off = rng.integers(-k_pos, k_pos + 1, b)
    98	    t2 = np.clip(t1 + off, 0, T)
    99	    neg = y == 0
   100	    # 负例：在 [0, T] 里相隔 ≥ k_neg 的位置中均匀取。左边可选 t1-k_neg+1 个，右边 T-t1-k_neg+1 个
   101	    left = np.maximum(t1 - k_neg + 1, 0)
   102	    right = np.maximum(T - t1 - k_neg + 1, 0)
   103	    tot = left + right
   104	    bad = neg & (tot == 0)
   105	    y[bad] = 1.0                       # 轨迹太短找不到负例时改成正例（T ≥ 2·k_neg 时不会发生）
   106	    neg = y == 0
   107	    u = (rng.random(b) * np.maximum(tot, 1)).astype(np.int64)
   108	    t2n = np.where(u < left, u, t1 + k_neg + (u - left))
   109	    t2 = np.where(neg, t2n, t2)
   110	    return i, t1, t2, y

   113	def sample_loc(ds: Dataset, rng: np.random.Generator, b: int, k_loc: int):
   114	    """返回 (i, t, k, a)：o_t 与 o_{t+k}，标签是 t 时刻的动作。"""
   115	    T = ds.horizon
   116	    i = rng.integers(0, ds.n_traj, b)
   117	    t = rng.integers(0, T, b)
   118	    k = 1 + (rng.random(b) * k_loc).astype(np.int64)
   119	    k = np.minimum(k, T - t)
   120	    return i, t, k, ds.act[i, t]

   169	def build_graph(p_pair: np.ndarray, keys: list[bytes], cfg: TopoPlanConfig) -> list[list[tuple[int, float, int]]]:
   170	    """p_pair：(n, n) 前缀各帧两两的 R 概率；keys：各帧观察指纹。返回有向带权邻接表 out[v] = [(u, 代价, 种类)]。"""
   171	    n = p_pair.shape[0]
   172	    out: list[list[tuple[int, float, int]]] = [[] for _ in range(n)]
   173	    for v in range(n - 1):
   174	        if keys[v] == keys[v + 1]:                   # 这一步观察没变（撞墙）
   175	            out[v].append((v + 1, cfg.w_same, SAME))
   176	            out[v + 1].append((v, cfg.w_same, SAME))
   177	        else:
   178	            out[v].append((v + 1, 1.0, FWD))
   179	            out[v + 1].append((v, cfg.w_rev, REV))
   180	    groups: dict[bytes, list[int]] = {}
   181	    for v, k in enumerate(keys):
   182	        groups.setdefault(k, []).append(v)
   183	    for vs in groups.values():
   184	        for a in vs:
   185	            for b in vs:
   186	                if a != b and abs(a - b) > 1:
   187	                    out[a].append((b, cfg.w_same, SAME))
   188	    if cfg.near:
   189	        gap = np.abs(np.arange(n)[:, None] - np.arange(n)[None])
   190	        ii, jj = np.nonzero((p_pair >= cfg.s_near) & (gap >= cfg.min_gap))
   191	        for i, j in zip(ii.tolist(), jj.tolist()):
   192	            if keys[i] != keys[j]:
   193	                out[i].append((j, cfg.w_near, NEAR))

--- topo.py: dist_to, hops, path_ahead ---
   197	def dist_to(out: list[list[tuple[int, float, int]]], targets) -> np.ndarray:
   198	    """Dijkstra：每个节点到目标集合的最小代价（到不了是 inf）。"""
   199	    import heapq
   200	    n = len(out)
   201	    rev: list[list[tuple[int, float]]] = [[] for _ in range(n)]
   202	    for v, es in enumerate(out):
   203	        for u, w, _ in es:
   204	            rev[u].append((v, w))
   205	    d = np.full(n, np.inf)
   206	    h = []
   207	    for s in targets:
   208	        d[s] = 0.0
   209	        h.append((0.0, int(s)))
   210	    heapq.heapify(h)
   211	    while h:
   212	        dv, v = heapq.heappop(h)
   213	        if dv > d[v]:
   214	            continue
   215	        for u, w in rev[v]:
   216	            if dv + w < d[u]:
   217	                d[u] = dv + w
   218	                heapq.heappush(h, (dv + w, u))
   219	    return d
   220	
   221	
   222	def hops(out, src: int, k: int) -> dict[int, int]:
   223	    """不分方向，src 的 k 跳以内的节点及跳数；同一处的边不计跳数（0-1 BFS）。"""
   224	    d = {src: 0}
   225	    q = deque([src])
   226	    while q:
   227	        v = q.popleft()
   228	        for u, _, kind in out[v]:
   229	            du = d[v] + (0 if kind == SAME else 1)
   230	            if du <= k and du < d.get(u, k + 1):
   231	                d[u] = du
   232	                if kind == SAME:
   233	                    q.appendleft(u)
   234	                else:
   235	                    q.append(u)
   236	    return d
   237

   239	def path_ahead(out, d: np.ndarray, cur: int, h: int) -> list[tuple[int, int]]:
   240	    """从 cur 沿最短路走最多 h 条边，返回 [(下一个节点, 边的种类)]。并列时取编号大的（确定性）。"""
   241	    path, v = [], cur
   242	    for _ in range(h):
   243	        if not np.isfinite(d[v]) or d[v] == 0:
   244	            break
   245	        best = None
   246	        for u, w, kind in out[v]:
   247	            c = w + d[u]
   248	            if c <= d[v] + 1e-9 and (best is None or (c, -u) < (best[0], -best[1])):
   249	                best = (c, u, kind)
   250	        if best is None:
   251	            break
   252	        path.append((best[1], best[2]))
   253	        v = best[1]
   254	    return path

   372	### 10.2 新组 latent_topo（SPTM / ViNG 式拓扑记忆，`src/ainerzero/v9/mini/topo.py`）
   373	
   374	这是文献里这类任务的标准做法：先给一段探索序列，再走到序列里见过的目标画面。全部自监督，只用训练游走的时间间隔和智能体自己的动作，不用真值位姿或距离。
   375	
   376	**学到的部分：**
   377	- **可达性网络 R(o_i, o_j)：** 同一条轨迹上相隔不超过 3 步为正例，至少 25 步为负例。
   378	- **运动网络 L(o_t, o_{t+k})：** k ≤ 4，预测 a_t。
   379	- **编码器：** 它有自己的，不共用世界模型。这是在给潜空间一方最好的机会，不是在检验共用编码器。
   380	
   381	**记忆图：**
   382	- 节点是前缀的每一帧。
   383	- 沿前缀正向走时，重放当时记下的动作。
   384	- 逆向的时间边，以及学到的捷径边（R ≥ 0.95、观察不同、时间上隔 ≥ 4），交给运动网络。
   385	- 观察完全相同的两帧算同一处，代价接近 0。
   386	
   387	**定位：** 先找观察完全相同的帧，取离预期节点最近的；找不到，就在预期节点 4 跳以内按 R 取最像的。
   388	
   389	**已知量：** 和地图组一样，知道任务从前缀终点出发，有指南针朝向。
   390	
   391	**探索中修掉的三处**（每处都先逐步跟踪查到原因再改）：
   392	1. **原地打转。** 路径的下一跳常常是同一位姿的另一帧，运动网络看到两帧一样的画面，给不出有用的动作。改为沿正向时间边重放动作，同一处不用动。
   393	2. **虫洞。** R 对同一房间里只差一格的两帧也给出 ≥ 0.95。「同一处」边代价为 0，串起来就成了长距离虫洞，例如 (8,15) 一步跳到 (15,14)。改为只有观察完全相同才算同一处；学到的捷径边按 2 步计代价。
   394	3. **定位邻域。** 原来按跳数算邻域，同一处的链也计了跳数，转身后正确的那一帧落在邻域外。改为同一处的边不计跳数（0-1 BFS）。

   424	### 10.5 读法
   425	
   426	1. **latent_topo 不是稻草人。** 短目标从随机水平（约 0.1）升到约 0.5。
   427	2. **长目标仍接近 0。** 局部控制换成真值最短路，m1 也只有 0.1 左右，m2 是 0。
   428	3. **原因是结构性的。**
   429	   - 以帧为节点的记忆，只记得走过的转移。
   430	   - 就算局部控制完美，m1 的上限也只有 0.45。
   431	   - 按格合并成地点、但前进只能沿走过的方向时，上限是 0.20。
   432	   - 要到 1.00，必须知道「从 A 往北走到 B，就能从 B 往南走回 A」，以及看见过、没站过的格也能走。这正是度量地图按设计写进去的几何。
   433	4. **H1 更准确的说法：** 显式几何（度量地图）胜过学到的潜空间记忆和拓扑记忆。差距主要来自记忆结构，不是我们的学习部件太弱。
   434	5. **还没覆盖的强对照：**
   435	   - 用注意力读整段前缀的长记忆（Transformer）；
   436	   - Dreamer 式的循环世界模型（gru_belief 只是很小的同类）。
