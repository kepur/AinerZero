import sys, time, random
from ainerzero.v9.experiments.s2 import data, world, spec

assert "final" not in sys.argv

def run_scan(partition, n_worlds):
    ns = spec.SEED_NAMESPACES[partition]
    redraw_counts = {}
    attempts_counts = {}
    t0 = time.time()
    for w in range(n_worlds):
        w_seed = data._get_derived_seed(ns, partition, "world", w)
        prep_rng = random.Random(data._get_derived_seed(ns, partition, "prep", w))
        layout, pairs, summary = world.prepare_world_with_long_pairs(w_seed, prep_rng, n_long_pairs=6)
        redraws = summary["world_redraws"]
        redraw_counts[w] = redraws
        attempts_counts[w] = redraws + 1
    elapsed = time.time() - t0
    max_w = max(attempts_counts, key=attempts_counts.get)
    print(f"{partition}: {n_worlds} worlds scanned in {elapsed:.2f}s, max attempts = {attempts_counts[max_w]} (world {max_w})")
    return attempts_counts

train_atts = run_scan("train", 200)
assert train_atts[11] == 8, f"Train world 11 expected 8 attempts, got {train_atts[11]}"
assert train_atts[173] == 20, f"Train world 173 expected 20 attempts, got {train_atts[173]}"
assert max(train_atts.values()) == 20

dev_atts = run_scan("dev", 50)
assert dev_atts[14] == 17, f"Dev world 14 expected 17 attempts, got {dev_atts[14]}"
assert max(dev_atts.values()) == 17

print("ALL SCANS PASSED SUCCESSFULLY!")
